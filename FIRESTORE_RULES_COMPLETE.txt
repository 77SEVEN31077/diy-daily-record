rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 舊 records collection — 完全禁止（不得 allow read, write: if true）
    match /records/{docId} {
      allow read, write: if false;
    }

    // 舊 messages collection — 已停用
    match /messages/{docId} {
      allow read, write: if false;
    }

    // 本月排行榜聚合資料（僅公開欄位，不含私人紀錄時間）
    match /monthlyLeaderboard/{docId} {
      allow read: if true;

      allow create: if isValidLeaderboardCreate();

      allow update: if isValidLeaderboardUpdate();

      allow delete: if false;
    }

    // 其餘路徑一律拒絕
    match /{document=**} {
      allow read, write: if false;
    }

    function isValidDisplayName(name) {
      return name is string && name.size() >= 1 && name.size() <= 24;
    }

    function isValidMonth(month) {
      return month is string && month.matches('^\\d{4}-\\d{2}$');
    }

    function leaderboardAllowedKeys() {
      return ['displayName', 'month', 'count', 'updatedAt'];
    }

    function isValidLeaderboardCreate() {
      let data = request.resource.data;
      return data.keys().hasOnly(leaderboardAllowedKeys())
        && isValidDisplayName(data.displayName)
        && isValidMonth(data.month)
        && data.count is int
        && data.count == 1
        && data.updatedAt is timestamp;
    }

    function isValidLeaderboardUpdate() {
      let data = request.resource.data;
      let prev = resource.data;
      return data.keys().hasOnly(leaderboardAllowedKeys())
        && isValidDisplayName(data.displayName)
        && data.month == prev.month
        && isValidMonth(data.month)
        && data.count is int
        && prev.count is int
        && data.updatedAt is timestamp
        && (
          (data.count == prev.count + 1 && data.updatedAt > prev.updatedAt)
          || (data.count == prev.count && data.displayName != prev.displayName && data.updatedAt >= prev.updatedAt)
        );
    }
  }
}
